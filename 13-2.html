核心代码

<!-- 引入 Lucide JS -->
<script src="cdn/lucide.js"></script>

<!-- 使用 data-lucide 属性 -->
<i data-lucide="home"></i>
<i data-lucide="heart"></i>
<i data-lucide="star"></i>

<!-- 自定义尺寸和颜色 -->
<i data-lucide="star" width="48" height="48" style="color: #e74c3c;"></i>

<!-- 自定义描边宽度 -->
<i data-lucide="star" stroke-width="3"></i>

<!-- 初始化图标 -->
<script>
    lucide.createIcons();
</script>
