核心代码

<!-- 引入 Font Awesome CSS -->
<link rel="stylesheet" href="cdn/fontawesome-all.min.css">

<!-- 使用字体图标 -->
<i class="fas fa-home"></i>
<i class="fas fa-heart"></i>
<i class="far fa-star"></i>
<i class="fab fa-github"></i>

<!-- 添加动画效果 -->
<i class="fas fa-cog fa-spin"></i>
<i class="fas fa-heart fa-beat"></i>
